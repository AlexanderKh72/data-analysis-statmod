```{r}
cols.punk <- c(#"popularity",
          #"duration_ms", 
          #"energy",
          "speechiness", 
          "acousticness", 
          #"instrumentalness", 
          "liveness")
raw.spotify |> 
  filter(track_genre == "classical", if_any(all_of(cols.punk), ~ . <= 1e-6)) |> select(all_of(cols.punk))

spotify.punk <- raw.spotify |> 
  mutate(across(all_of(cols.punk), log, .names = "{col}.log")) |> 
    mutate(calmness.log = log(1 - energy)) |>
  select(-all_of(cols.punk), -energy)
```

```{r}
active.cols.punk <- spotify.punk |> 
            select(where(is.numeric))|>
                select(-X, -key, -time_signature, -instrumentalness) |> colnames()
active.spotify.punk <- spotify.punk |> filter(track_genre == "punk")
res.pca.punk <- PCA(active.spotify.punk |> 
            select(all_of(active.cols.punk)), graph = FALSE, ncp=10)
```

```{r}
get_eigenvalue(res.pca.punk)
fviz_eig(res.pca.punk, addlabels = TRUE)
```

Такая же ситуация с собственными числами, как и в поп-музыке

```{r}
res.pca.var <- get_pca_var(res.pca.punk)
t(t(res.pca.var$coord) / sqrt(get_eigenvalue(res.pca.punk)[,"eigenvalue"]))
t(t(res.pca.var$coord) / sqrt(get_eigenvalue(res.pca.punk)[,"eigenvalue"])) |> corrplot(col = rev(COL2('RdBu')))
```
